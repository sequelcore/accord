/// How one proof of a file ended.
pub enum ProofOutcome { Passed, Failed, Vacuous, TimedOut, Cancelled, NotStarted }

intent "A proof passes only when nothing stopped it and something was verified."

pub fn proof_outcome(cancelled: Bool, timed_out: Bool, started: Bool,
                     regenerated: Bool, check_exit: i32, verified: i64) -> ProofOutcome
  requires verified >= 0
  ensures result == Passed <==> !cancelled && !timed_out && started
                                 && regenerated && check_exit == 0 && verified > 0
  ensures result == Vacuous <==> !cancelled && !timed_out && started
                                  && regenerated && check_exit == 0 && verified == 0
  ensures result == NotStarted <==> !cancelled && !timed_out && !started
  ensures cancelled ==> result == Cancelled
  ensures !cancelled && timed_out ==> result == TimedOut
  ensures result == Failed ==> started && (!regenerated || check_exit != 0)
{
  if cancelled { return Cancelled }
  if timed_out { return TimedOut }
  if !started { return NotStarted }
  if !regenerated || check_exit != 0 { return Failed }
  if verified > 0 { Passed } else { Vacuous }
}

example proof_outcome(cancelled: false, timed_out: false, started: true,
                      regenerated: true, check_exit: 0, verified: 3) == Passed
