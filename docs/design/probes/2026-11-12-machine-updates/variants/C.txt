/// Where a retried model call stands.
enum RetryPhase { Failed, Waiting, Calling, Ended }

/// A failed model call, retried within a budget and a time window.
machine ProviderRetry {
    intent "Retry a failed model call at most three times within its window, wait before each retry, and never call again after cancellation."

    state {
        phase: RetryPhase,
        time: i32,
        due: i32,
        retries: i32,
        calls: i32,
        stopped: Bool,
        calls_at_stop: i32,
        terminal: Bool,
        last_start: i32,
        last_due: i32,
    }

    init start(is_terminal: Bool) {
        phase = Failed
        time = 0
        due = 0
        retries = 0
        calls = 1
        stopped = false
        calls_at_stop = 0
        terminal = is_terminal
        last_start = 0
        last_due = 0
    }

    event schedule(delay: i32)
        requires phase == Failed
        requires delay == 1 or delay == 60 or delay == 121
    {
        if not terminal and not stopped and retries < 3 and time + delay < 120 {
            phase' = Waiting
            due' = time + delay
            retries' = retries + 1
        } else {
            phase' = Ended
        }
    }

    event advance(to: i32)
        requires phase == Waiting and time < 120
        requires (to == due or to == 120) and to >= time
    {
        time' = to
    }

    event cancel()
        requires not stopped and phase != Ended
    {
        stopped' = true
        calls_at_stop' = calls
    }

    event start_call()
        requires phase == Waiting and time >= due and time < 120 and not stopped
    {
        phase' = Calling
        calls' = calls + 1
        last_start' = time
        last_due' = due
    }

    event failure(is_terminal: Bool)
        requires phase == Calling
    {
        phase' = Failed
        terminal' = is_terminal
    }

    event success()
        requires phase == Calling
    {
        phase' = Ended
    }

    invariant bounds: retries <= 3 and calls <= 4 and last_start < 120
    invariant cancel_stops_calls: stopped implies calls == calls_at_stop
    invariant wait_respected: last_start >= last_due

    helper invariant calls_follow_retries: calls <= retries + 1
    helper invariant waiting_before_call: phase == Waiting implies calls <= retries
    helper invariant time_in_window: time <= 120
    helper invariant due_in_window: due <= 120
}
