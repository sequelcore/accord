/// The colors of a traffic light.
pub enum Light { Red, Yellow, Green }

pub fn next(light: Light) -> Light
    intent "Cycles a traffic light."
decide {
    | light == Red    => Green
    | light == Green  => Yellow
    | light == Yellow => Red
}

pub fn after_two(light: Light) -> Light {
    next(next(light))
}

example after_two(Light.Red) == Yellow
