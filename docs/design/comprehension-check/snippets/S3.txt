/// Adds two non-negative counts.
pub fn total(a: i32, b: i32) -> i32
    intent "Adds two counts."
    requires a >= 0 and b >= 0
    requires a + b <= 2147483647
    ensures result == a + b
{
    a + b
}

example total(a: 2, b: 3) == 5
