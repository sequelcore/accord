/// The colors of a traffic light.
pub enum Light { Red, Yellow, Green }

/// The alert a crossing shows.
pub enum Alert { Red, Clear }

pub fn next(light: Light) -> Light
    intent "Cycles a traffic light."
decide {
    | light == Light.Red    => Light.Green
    | light == Light.Green  => Light.Yellow
    | light == Light.Yellow => Light.Red
}

pub fn alert_for(light: Light) -> Alert
    intent "Shows the red alert while the light is red."
decide {
    | light == Light.Red => Alert.Red
    | otherwise          => Alert.Clear
}

example next(Light.Green) == Light.Yellow
example alert_for(next(Light.Yellow)) == Alert.Red
