/// A grade for an exam score.
pub enum Grade { Fail, Pass, Honors }

pub fn grade(score: i32) -> Grade
    intent "Scores of 90 and above earn honors; 60 and above pass."
    requires score >= 0 and score <= 100
decide {
    | score >= 60 => Grade.Pass
    | score >= 90 => Grade.Honors
    | otherwise   => Grade.Fail
}
