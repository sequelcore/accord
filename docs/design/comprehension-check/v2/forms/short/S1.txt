/// What a library charges for a late return.
pub enum Fee { Waived, Flat, PerDay }

pub fn late_fee(days_late: i32, member: Bool, first_offense: Bool) -> Fee
    intent "Members never pay for their first late return."
    requires days_late >= 0
decide {
    | days_late == 0           => Waived
    | member and first_offense => Waived
    | days_late <= 3           => Flat
    | otherwise                => PerDay
}

example late_fee(days_late: 5, member: true, first_offense: true) == Waived
